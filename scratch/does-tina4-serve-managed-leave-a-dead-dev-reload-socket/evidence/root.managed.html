Getting Started

// index.php
require_once 'vendor/autoload.php';

$app = new \Tina4\App();

\Tina4\Router::get('/hello', function ($request, $response) {
    return $response->json(['message' => 'Hello World!']);
});

$app->run();  // starts on port 7145
